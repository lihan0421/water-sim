**Goal:** 构建浏览器端实时水体仿真：大海/河流/水槽三场景，FFT 海浪 + 交互波动方程双层高度场，浮力刚体（可驾驶船、投掷物），浪花泡沫。

```ts
// src/core/App.ts
import * as THREE from 'three/webgpu';
import GUI from 'lil-gui';
import type { WaterScene, SceneContext } from './WaterScene';

export class App {
  private renderer: THREE.WebGPURenderer;
  private camera: THREE.PerspectiveCamera;
  private gui = new GUI({ title: 'Water Sim' });
  private sceneGui!: GUI;
  private current: WaterScene | null = null;
  private factories: Record<string, () => WaterScene>;
  private clock = new THREE.Clock();
  private fpsEl: HTMLDivElement;

  constructor(renderer: THREE.WebGPURenderer, factories: Record<string, () => WaterScene>) {
    this.renderer = renderer;
    this.factories = factories;
    this.camera = new THREE.PerspectiveCamera(60, innerWidth / innerHeight, 0.1, 4000);
    addEventListener('resize', () => {
      this.camera.aspect = innerWidth / innerHeight;
      this.camera.updateProjectionMatrix();
      renderer.setSize(innerWidth, innerHeight);
    });
    this.fpsEl = document.createElement('div');
    this.fpsEl.style.cssText = 'position:fixed;left:8px;top:8px;color:#0f0;font:12px monospace;z-index:9';
    document.body.appendChild(this.fpsEl);
    const names = Object.keys(factories);
    const state = { scene: names[0] };
    this.gui.add(state, 'scene', names).name('场景').onChange((n: string) => this.switchTo(n));
  }

  async switchTo(name: string) {
    this.current?.dispose();
    this.sceneGui?.destroy();
    this.sceneGui = this.gui.addFolder(name);
    this.current = this.factories[name]();
    const ctx: SceneContext = { renderer: this.renderer, camera: this.camera, gui: this.sceneGui, domElement: this.renderer.domElement };
    await this.current.init(ctx);
  }

  start() {
    let acc = 0, frames = 0;
    this.renderer.setAnimationLoop(() => {
      const dt = Math.min(this.clock.getDelta(), 1 / 20); // 防卡顿大步长
      acc += dt; frames++;
      if (acc > 0.5) { this.fpsEl.textContent = `${Math.round(frames / acc)} fps`; acc = 0; frames = 0; }
      if (this.current) {
        this.current.update(dt, this.clock.elapsedTime);
        this.renderer.render(this.current.scene, this.camera);
      }
    });
  }
}
```

```ts
// src/sim/fft/spectrum.ts
const G = 9.81;

export interface SpectrumParams {
  windSpeed: number;      // m/s
  windDirection: number;  // 弧度，0 = +X
  fetch: number;          // 风区长度 m
  amplitudeScale: number; // 总体幅度缩放（GUI 浪高）
}

export function peakOmega(U: number, F: number): number {
  return 22 * Math.pow((G * G) / (U * F), 1 / 3);
}

/** JONSWAP 频率谱 S(ω) */
export function jonswap(omega: number, U: number, F: number): number {
  if (omega <= 0) return 0;
  const wp = peakOmega(U, F);
  const alpha = 0.076 * Math.pow((U * U) / (F * G), 0.22);
  const gamma = 3.3;
  const sigma = omega <= wp ? 0.07 : 0.09;
  const r = Math.exp(-((omega - wp) ** 2) / (2 * sigma * sigma * wp * wp));
  return ((alpha * G * G) / omega ** 5) * Math.exp(-1.25 * (wp / omega) ** 4) * Math.pow(gamma, r);
}

/** cos² 方向扩散，归一化 ∫D dθ = 1，逆风留 2% 防止完全死寂 */
export function directionalSpread(theta: number, windDir: number): number {
  let d = theta - windDir;
  while (d > Math.PI) d -= 2 * Math.PI;
  while (d < -Math.PI) d += 2 * Math.PI;
  const forward = Math.abs(d) < Math.PI / 2 ? (2 / Math.PI) * Math.cos(d) ** 2 : 0;
  return forward * 0.98 + 0.02 / (2 * Math.PI);
}

/**
 * 初始频谱 h0。布局：N×N 网格，像素 (ix,iz) 对应波矢
 * k = 2π/L * (ix - N/2, iz - N/2)。每像素 4 个 float：
 * [h0(k).re, h0(k).im, h0(-k)*.re, h0(-k)*.im]，同时输出 ω(k)。
 * rng: 返回 [0,1) 的随机源（测试时可注入定值）。
 */
export function generateInitialSpectrum(
  N: number, L: number, p: SpectrumParams, rng: () => number = Math.random,
): { h0: Float32Array; omega: Float32Array } {
  const h0 = new Float32Array(N * N * 4);
  const omega = new Float32Array(N * N);
  const dk = (2 * Math.PI) / L;
  // 高斯随机（Box-Muller）
  const gauss = () => {
    const u1 = Math.max(rng(), 1e-9), u2 = rng();
    return Math.sqrt(-2 * Math.log(u1)) * Math.cos(2 * Math.PI * u2);
  };
  const amp = (kx: number, kz: number) => {
    const k = Math.hypot(kx, kz);
    if (k < 1e-6) return 0;
    const w = Math.sqrt(G * k);            // 深水色散
    const theta = Math.atan2(kz, kx);
    // S(k,θ) = S(ω)·D(θ)·dω/dk / k，dω/dk = g/(2ω)
    const Sk = (jonswap(w, p.windSpeed, p.fetch) * directionalSpread(theta, p.windDirection) * (G / (2 * w))) / k;
    return Math.sqrt(2 * Sk) * dk * p.amplitudeScale; // |h0| = √(2 S Δk²) / √2 · √2 → √(2S)·Δk
  };
  for (let iz = 0; iz < N; iz++) {
    for (let ix = 0; ix < N; ix++) {
      const kx = dk * (ix - N / 2), kz = dk * (iz - N / 2);
      const i = iz * N + ix;
      omega[i] = Math.sqrt(G * Math.hypot(kx, kz));
      const a = amp(kx, kz), b = amp(-kx, -kz);
      const inv = Math.SQRT1_2;
      h0[i * 4 + 0] = inv * gauss() * a;
      h0[i * 4 + 1] = inv * gauss() * a;
      h0[i * 4 + 2] = inv * gauss() * b;   // h0(-k) 的实部
      h0[i * 4 + 3] = -inv * gauss() * b;  // 共轭 → 虚部取负
    }
  }
  return { h0, omega };
}
```

### Task 4: FFT 蝶形预计算 + CPU 参考 IFFT（TDD）

GPU FFT 用经典「蝶形查找表 + ping-pong」法：CPU 预计算每个 stage 的索引与旋转因子，GPU 每 pass 只做一次乘加。本任务先在 CPU 上把查找表算法验证正确。

```ts
/** 直接逆 DFT 参考：x[n] = Σ X[k]·e^{+2πi kn/N}（不除 N，与 GPU 实现一致） */
function directIdft(re: Float32Array, im: Float32Array): { re: Float32Array; im: Float32Array } {
  const N = re.length;
  const or_ = new Float32Array(N), oi = new Float32Array(N);
  for (let n = 0; n < N; n++) {
    for (let k = 0; k < N; k++) {
      const ph = (2 * Math.PI * k * n) / N;
      or_[n] += re[k] * Math.cos(ph) - im[k] * Math.sin(ph);
      oi[n] += re[k] * Math.sin(ph) + im[k] * Math.cos(ph);
    }
  }
  return { re: or_, im: oi };
}

describe('butterfly FFT', () => {
  it('bitReverse 正确', () => {
    expect(bitReverse(1, 3)).toBe(4);
    expect(bitReverse(3, 3)).toBe(6);
```

```ts
// src/sim/fft/butterfly.ts

export function bitReverse(i: number, bits: number): number {
  let r = 0;
  for (let b = 0; b < bits; b++) { r = (r << 1) | (i & 1); i >>= 1; }
  return r;
}

/**
 * 蝶形查找表：stages × N 个条目，每条 4 float：
 * [twiddle.re, twiddle.im, indexA, indexB]
 * 任意 wing 统一公式 out[i] = in[A] + W·in[B]（逆变换 W = e^{+2πik/N}）。
 */
export function createButterflyData(N: number): Float32Array {
  const stages = Math.log2(N);
  const data = new Float32Array(stages * N * 4);
  for (let s = 0; s < stages; s++) {
    const span = 1 << s;            // 蝶形半宽
    const seg = span << 1;          // 蝶形全宽
    for (let i = 0; i < N; i++) {
      const k = ((i * N) / seg) % N;            // 旋转因子序号（下翼自动相差 N/2 → 取负）
      const wr = Math.cos((2 * Math.PI * k) / N);
      const wi = Math.sin((2 * Math.PI * k) / N); // 逆变换取 +sin
      const top = i % seg < span;
      let a: number, b: number;
      if (s === 0) { a = top ? bitReverse(i, stages) : bitReverse(i - 1, stages);
                     b = top ? bitReverse(i + 1, stages) : bitReverse(i, stages); }
      else         { a = top ? i : i - span;  b = top ? i + span : i; }
      const o = (s * N + i) * 4;
      data[o] = wr; data[o + 1] = wi; data[o + 2] = a; data[o + 3] = b;
    }
  }
  return data;
}

/** CPU 参考：用蝶形表跑 1D IFFT（与 GPU kernel 逐 pass 等价，用于测试） */
export function ifft1DWithButterfly(re: Float32Array, im: Float32Array, bf: Float32Array, N: number) {
  const stages = Math.log2(N);
  let curR = Float32Array.from(re), curI = Float32Array.from(im);
  for (let s = 0; s < stages; s++) {
    const nxtR = new Float32Array(N), nxtI = new Float32Array(N);
    for (let i = 0; i < N; i++) {
      const o = (s * N + i) * 4;
      const wr = bf[o], wi = bf[o + 1], a = bf[o + 2], b = bf[o + 3];
      nxtR[i] = curR[a] + wr * curR[b] - wi * curI[b];
      nxtI[i] = curI[a] + wr * curI[b] + wi * curR[b];
    }
    curR = nxtR; curI = nxtI;
  }
  return { re: curR, im: curI };
}
```

### Task 5: GPU FFT 海浪管线（FFTWaves）

把 Task 3/4 的数据搬上 GPU：每帧「频谱时变 → 2×(横向 log₂N + 纵向 log₂N) IFFT pass → 置换/导出」，输出位移与法线/泡沫 StorageTexture。每级联 2 路 IFFT：①高度 h；②打包 Dx + i·Dz（一次复 FFT 解出两个实序列）。

```ts
// src/sim/fft/FFTWaves.ts
import * as THREE from 'three/webgpu';
import {
  Fn, instancedArray, attributeArray, instanceIndex, uniform,
  float, int, vec2, vec4, ivec2, textureStore, textureLoad,
} from 'three/tsl';
import { generateInitialSpectrum, type SpectrumParams } from './spectrum';
import { createButterflyData } from './butterfly';

export interface CascadeConfig { domainSize: number } // 米

export class FFTCascade {
  readonly displacementTex: THREE.StorageTexture;  // rgba16f: xyz=位移, w=高度备份
  readonly normalFoamTex: THREE.StorageTexture;    // rgba16f: xyz=法线, w=泡沫(Jacobian)
  readonly heightBuffer: ReturnType<typeof instancedArray>; // 供物理回读：vec2(高度, 0)…用 float
  private computes: { update: any; passes: any[]; output: any; normals: any };
  private timeU = uniform(0);
  private choppyU: ReturnType<typeof uniform>;
  readonly domainSize: number;

  constructor(private N: number, cfg: CascadeConfig, params: SpectrumParams, choppyU: any) {
    this.domainSize = cfg.domainSize;
    this.choppyU = choppyU;
    const { h0, omega } = generateInitialSpectrum(N, cfg.domainSize, params);
    // 静态数据
    const h0Buf = attributeArray(h0, 'vec4');
    const omegaBuf = attributeArray(omega, 'float');
    const bfData = createButterflyData(N);
    const bfBuf = attributeArray(bfData, 'vec4');
    // ping-pong 复数缓冲：每元素 vec4 = [h.re, h.im, D.re, D.im]（h 与打包位移并行算）
    const pingA = instancedArray(N * N, 'vec4');
    const pingB = instancedArray(N * N, 'vec4');
    this.heightBuffer = instancedArray(N * N, 'float');

    this.displacementTex = makeStorageTex(N);
    this.normalFoamTex = makeStorageTex(N);

    const NN = N * N, log2N = Math.log2(N);
    const Ni = int(N);

    // Pass 0: 频谱时变 h(k,t) = h0(k)e^{iωt} + h0(-k)* e^{-iωt}；位移谱 = -i k̂ ·h 打包 Dx+iDz
    const dk = (2 * Math.PI) / cfg.domainSize;
    const update = Fn(() => {
      const i = instanceIndex;
      const ix = i.mod(Ni), iz = i.div(Ni);
      const h0v = h0Buf.element(i);
      const w = omegaBuf.element(i);
      const c = w.mul(this.timeU).cos(), s = w.mul(this.timeU).sin();
      // h = h0·e^{iwt} + h0c·e^{-iwt}
      const hr = h0v.x.mul(c).sub(h0v.y.mul(s)).add(h0v.z.mul(c).add(h0v.w.mul(s)));
      const hi = h0v.x.mul(s).add(h0v.y.mul(c)).add(h0v.w.mul(c).sub(h0v.z.mul(s)));
      // k 单位向量
      const kx = float(ix).sub(N / 2).mul(dk), kz = float(iz).sub(N / 2).mul(dk);
      const klen = vec2(kx, kz).length().max(1e-6);
      const nx = kx.div(klen), nz = kz.div(klen);
      // D = -i k̂ h ; 打包 P = Dx + i·Dz：
      // Dx = (hi·nx, -hr·nx), Dz = (hi·nz, -hr·nz) → P.re = hi·nx + hr·nz, P.im = -hr·nx + hi·nz
      const pr = hi.mul(nx).add(hr.mul(nz));
      const pi = hr.negate().mul(nx).add(hi.mul(nz));
      pingA.element(i).assign(vec4(hr, hi, pr, pi));
    })().compute(NN);

    // 蝶形 pass：dir=0 横向（行内 FFT），dir=1 纵向；每 stage 一个 compute 节点
    const passes: any[] = [];
    let src = pingA, dst = pingB;
    for (let dir = 0; dir < 2; dir++) {
      for (let s = 0; s < log2N; s++) {
        const sConst = s, dirConst = dir, S = src, D = dst;
        passes.push(Fn(() => {
          const i = instanceIndex;
          const ix = i.mod(Ni), iz = i.div(Ni);
          const lane = dirConst === 0 ? ix : iz;       // 当前 FFT 维度内坐标
          const bf = bfBuf.element(int(sConst * N).add(lane));
          const a = bf.z.toInt(), b = bf.w.toInt();
          const idxA = dirConst === 0 ? iz.mul(Ni).add(a) : a.mul(Ni).add(ix);
          const idxB = dirConst === 0 ? iz.mul(Ni).add(b) : b.mul(Ni).add(ix);
          const va = S.element(idxA), vb = S.element(idxB);
          const wr = bf.x, wi = bf.y;
          // 两路复数同时蝶形
          D.element(i).assign(vec4(
            va.x.add(wr.mul(vb.x)).sub(wi.mul(vb.y)),
            va.y.add(wr.mul(vb.y)).add(wi.mul(vb.x)),
            va.z.add(wr.mul(vb.z)).sub(wi.mul(vb.w)),
            va.w.add(wr.mul(vb.w)).add(wi.mul(vb.z)),
          ));
        })().compute(NN));
        [src, dst] = [dst, src];
      }
    }
    const finalBuf = src; // 偶数次交换后最终结果所在缓冲

    // 输出 pass：(-1)^(x+y) 置换 + 写位移纹理 + 高度缓冲
    const output = Fn(() => {
      const i = instanceIndex;
      const ix = i.mod(Ni), iz = i.div(Ni);
      const sign = float(1).sub(ix.add(iz).mod(2).toFloat().mul(2)); // (-1)^(x+z)
      const v = finalBuf.element(i);
      const h = v.x.mul(sign);
      const dx = v.z.mul(sign).mul(this.choppyU);
      const dz = v.w.mul(sign).mul(this.choppyU);
      textureStore(this.displacementTex, ivec2(ix, iz), vec4(dx, h, dz, h));
      this.heightBuffer.element(i).assign(h);
    })().compute(NN);

    // 法线 + Jacobian 泡沫 pass（位移纹理有限差分）
    const texel = cfg.domainSize / N;
    const normals = Fn(() => {
      const i = instanceIndex;
      const ix = i.mod(Ni), iz = i.div(Ni);
      const wrap = (v: any) => v.add(Ni).mod(Ni);
      const ld = (x: any, z: any) => this.displacementTexLoad(ivec2(wrap(x), wrap(z)));
      const r = ld(ix.add(1), iz), l = ld(ix.sub(1), iz);
      const t = ld(ix, iz.add(1)), b = ld(ix, iz.sub(1));
      const dhdx = r.y.sub(l.y).div(2 * texel), dhdz = t.y.sub(b.y).div(2 * texel);
      const n = vec4(dhdx.negate(), 1, dhdz.negate(), 0).normalize();
      // Jacobian
      const jxx = float(1).add(r.x.sub(l.x).div(2 * texel));
      const jzz = float(1).add(t.z.sub(b.z).div(2 * texel));
      const jxz = t.x.sub(b.x).div(2 * texel);
      const jac = jxx.mul(jzz).sub(jxz.mul(jxz));
      const foam = float(0.6).sub(jac).max(0); // J < 0.6 起沫
      textureStore(this.normalFoamTex, ivec2(ix, iz), vec4(n.x, n.y, n.z, foam));
    })().compute(NN);

    this.computes = { update, passes, output, normals };
  }

  // displacementTex 的 texelFetch 读取。three r180+ 从 'three/tsl' 导入 textureLoad；
  // 若所装版本无此导出，改用 texture(tex).load(coord)（查官方 webgpu_compute_texture 示例确认）
  private displacementTexLoad(coord: any) {
    return textureLoad(this.displacementTex, coord);
  }

  update(renderer: THREE.WebGPURenderer, time: number) {
    this.timeU.value = time;
    renderer.compute(this.computes.update);
    for (const p of this.computes.passes) renderer.compute(p);
    renderer.compute(this.computes.output);
    renderer.compute(this.computes.normals);
  }
}

function makeStorageTex(N: number) {
  const t = new THREE.StorageTexture(N, N);
  t.type = THREE.HalfFloatType;
  t.format = THREE.RGBAFormat;
  t.magFilter = t.minFilter = THREE.LinearFilter;
  t.wrapS = t.wrapT = THREE.RepeatWrapping;
  t.generateMipmaps = false;
  return t;
}

/** 三级联封装 */
export class FFTWaves {
  cascades: FFTCascade[];
  choppyU = uniform(1.2);
  constructor(N: number, params: SpectrumParams, domains = [250, 60, 15]) {
    this.cascades = domains.map((d) => new FFTCascade(N, { domainSize: d }, params, this.choppyU));
  }
  update(renderer: THREE.WebGPURenderer, time: number) {
    for (const c of this.cascades) c.update(renderer, time);
  }
}
```

注意（执行者必读）：

```ts
// src/water/WaterMaterial.ts
import * as THREE from 'three/webgpu';
import {
  Fn, texture, uniform, vec2, vec3, vec4, float, positionLocal, positionWorld,
  normalize, mix, max, dot, pow, reflect, cameraPosition, exp, clamp,
} from 'three/tsl';
import { skyColor } from '../ui/skyNode';
import type { FFTWaves } from '../sim/fft/FFTWaves';

export interface InteractiveMaps {
  heightTex: THREE.StorageTexture;   // r=高度
  foamTex: THREE.StorageTexture;     // r=泡沫
  origin: ReturnType<typeof uniform>; // vec2 网格世界原点（中心）
  sizeMeters: number;
}

export interface WaterMaterialOpts {
  fft?: FFTWaves | null;
  interactive?: InteractiveMaps | null;
  deepColor?: THREE.Color;
  shallowColor?: THREE.Color;
}

export function createWaterMaterial(o: WaterMaterialOpts) {
  const mat = new THREE.MeshBasicNodeMaterial(); // 光照全部手写，Basic 即可
  const deep = uniform(new THREE.Color(o.deepColor ?? new THREE.Color(0x06283d)));
  const shallow = uniform(new THREE.Color(o.shallowColor ?? new THREE.Color(0x1a7a8a)));
  const fftFade = uniform(1); // 河流场景调低

  // —— 位移（顶点）——
  const sampleInteractiveH = (wp: any) => {
    if (!o.interactive) return float(0);
    const i = o.interactive;
    const uvI = wp.xz.sub(i.origin).div(i.sizeMeters).add(0.5);
    const inside = uvI.x.greaterThan(0.002).and(uvI.x.lessThan(0.998))
      .and(uvI.y.greaterThan(0.002)).and(uvI.y.lessThan(0.998));
    return texture(i.heightTex, uvI).r.mul(inside.select(1, 0));
  };

  const displacedPos = Fn(() => {
    const wp = positionWorld;
    let disp: any = vec3(0);
    if (o.fft) for (const c of o.fft.cascades) {
      const uvC = wp.xz.div(c.domainSize);
      disp = disp.add(texture(c.displacementTex, uvC).xyz.mul(fftFade));
    }
    disp = disp.add(vec3(0, sampleInteractiveH(wp), 0));
    return positionLocal.add(disp);
  })();
  mat.positionNode = displacedPos;

  // —— 着色（片元）——
  mat.colorNode = Fn(() => {
    const wp = positionWorld;
    // 法线：FFT 级联法线混合 + 交互层差分
    let n: any = vec3(0, 1, 0);
    let foam: any = float(0);
    if (o.fft) for (const c of o.fft.cascades) {
      const nf = texture(c.normalFoamTex, wp.xz.div(c.domainSize));
      // 多级联法线按斜率叠加（x/z 分量相加，y 归一）
      n = normalize(vec3(n.x.add(nf.x.mul(fftFade)), 1, n.z.add(nf.z.mul(fftFade))));
      foam = foam.add(nf.w.mul(fftFade));
    }
    if (o.interactive) {
      const i = o.interactive;
      const uvI = wp.xz.sub(i.origin).div(i.sizeMeters).add(0.5);
      const e = 1.5 / i.sizeMeters;
      const hl = texture(i.heightTex, uvI.sub(vec2(e, 0))).r, hr = texture(i.heightTex, uvI.add(vec2(e, 0))).r;
      const hb = texture(i.heightTex, uvI.sub(vec2(0, e))).r, ht = texture(i.heightTex, uvI.add(vec2(0, e))).r;
      n = normalize(vec3(n.x.add(hl.sub(hr).mul(2)), 1, n.z.add(hb.sub(ht).mul(2))));
      foam = foam.add(texture(i.foamTex, uvI).r);
    }
    foam = clamp(foam, 0, 1);

    const V = normalize(cameraPosition.sub(wp));
    const fres = float(0.02).add(float(0.98).mul(pow(float(1).sub(max(dot(n, V), 0)), 5)));
    const refl = skyColor(reflect(V.negate(), n));
    // 体色：浪高处偏散射绿
    const heightTint = clamp(wp.y.mul(0.15).add(0.2), 0, 1);
    const body = mix(deep, shallow, heightTint);
    // 太阳高光
    const sun = normalize(vec3(0.4, 0.35, 0.6));
    const spec = pow(max(dot(n, normalize(V.add(sun))), 0), 360).mul(2);
    let col = mix(body, refl, fres).add(vec3(1, 0.95, 0.85).mul(spec));
    col = mix(col, vec3(0.95), clamp(foam.mul(1.4), 0, 1)); // 泡沫盖白
    return vec4(col, 1);
  })();
  return { material: mat, uniforms: { fftFade, deep, shallow } };
}
```

```ts
// src/sim/interactive/waveKernel.ts
export interface WaveParams {
  c2dt2: number;     // c²·dt²/dx²，稳定性要求 < 0.5
  damping: number;   // 每步速度衰减系数
  boundary: 'absorb' | 'reflect';
  flowX: number; flowZ: number; // 平流速度（格/秒），河流用
  dt: number;
}

/** 一步显式波动方程。next = 2h - prev·(1-damp) … + c²dt²∇²h，边缘按 boundary 处理 */
export function waveStep(prev: Float32Array, curr: Float32Array, next: Float32Array, N: number, p: WaveParams) {
  const at = (x: number, z: number) => {
    if (x < 0 || x >= N || z < 0 || z >= N) return 0;
    return curr[z * N + x];
  };
  for (let z = 0; z < N; z++) {
    for (let x = 0; x < N; x++) {
      const i = z * N + x;
      const edge = x === 0 || x === N - 1 || z === 0 || z === N - 1;
      if (edge) {
        next[i] = p.boundary === 'reflect'
          ? at(Math.min(Math.max(x, 1), N - 2), Math.min(Math.max(z, 1), N - 2)) // Neumann：复制内邻
          : curr[i] * 0.5;  // absorb：强阻尼吸收
        continue;
      }
      // 平流（半拉格朗日，双线性回溯）
      let h = curr[i], hp = prev[i];
      if (p.flowX !== 0 || p.flowZ !== 0) {
        const sx = x - p.flowX * p.dt, sz = z - p.flowZ * p.dt;
        const x0 = Math.floor(sx), z0 = Math.floor(sz), fx = sx - x0, fz = sz - z0;
        const bil = (a: Float32Array) => {
          const g = (xx: number, zz: number) => (xx < 0 || xx >= N || zz < 0 || zz >= N) ? 0 : a[zz * N + xx];
          return g(x0, z0) * (1 - fx) * (1 - fz) + g(x0 + 1, z0) * fx * (1 - fz)
               + g(x0, z0 + 1) * (1 - fx) * fz + g(x0 + 1, z0 + 1) * fx * fz;
        };
        h = bil(curr); hp = bil(prev);
      }
      const lap = at(x + 1, z) + at(x - 1, z) + at(x, z + 1) + at(x, z - 1) - 4 * h;
      const vel = (h - hp) * (1 - p.damping);
      next[i] = h + vel + p.c2dt2 * lap;
    }
  }
}
```

```ts
// src/sim/interactive/InteractiveWaves.ts
import * as THREE from 'three/webgpu';
import {
  Fn, instancedArray, instanceIndex, uniform, uniformArray, float, int, vec2, vec4,
  ivec2, textureStore, If, Loop, clamp, max,
} from 'three/tsl';

export interface InteractiveOpts {
  N?: number;             // 网格分辨率，默认 512
  sizeMeters: number;     // 网格覆盖边长
  boundary: 'absorb' | 'reflect';
  waveSpeed?: number;     // m/s，默认 6
  damping?: number;       // 默认 0.015
}

const MAX_DISTURB = 16;

export class InteractiveWaves {
  readonly N: number;
  readonly sizeMeters: number;
  readonly heightTex: THREE.StorageTexture;
  readonly foamTex: THREE.StorageTexture;
  readonly originU = uniform(new THREE.Vector2(0, 0)); // 网格中心世界坐标
  readonly heightBuffers: any[]; // ping-pong instancedArray，[0] 恒为「当前」供回读
  private computes: { step: any[]; export: any };
  private disturbPos = new Float32Array(MAX_DISTURB * 2);
  private disturbVal = new Float32Array(MAX_DISTURB * 2); // [强度, 半径(格)]
  private disturbCount = 0;
  private dPosU: any; private dValU: any; private dCountU = uniform(0);
  private c2dt2U = uniform(0.2); private dampU: any; private flowU = uniform(new THREE.Vector2(0, 0));
  private boundaryReflect: number;
  // 物理参数
  private waveSpeed: number; private dx: number;

  constructor(o: InteractiveOpts) {
    this.N = o.N ?? 512;
    this.sizeMeters = o.sizeMeters;
    this.dx = o.sizeMeters / this.N;
    this.waveSpeed = o.waveSpeed ?? 6;
    this.dampU = uniform(o.damping ?? 0.015);
    this.boundaryReflect = o.boundary === 'reflect' ? 1 : 0;
    const NN = this.N * this.N;
    const a = instancedArray(NN, 'float'), b = instancedArray(NN, 'float'), c = instancedArray(NN, 'float');
    this.heightBuffers = [a, b, c]; // prev, curr, next 轮换
    const foamBuf = instancedArray(NN, 'float');
    this.heightTex = makeTex(this.N); this.foamTex = makeTex(this.N);
    this.dPosU = uniformArray(Array.from({ length: MAX_DISTURB }, () => new THREE.Vector2()));
    this.dValU = uniformArray(Array.from({ length: MAX_DISTURB }, () => new THREE.Vector2()));

    const Ni = int(this.N);
    const reflectB = this.boundaryReflect;
    // 三个轮换排列各建一个 step compute（prev,curr,next 的三种角色分配）
    const mkStep = (prev: any, curr: any, next: any) => Fn(() => {
      const i = instanceIndex;
      const x = i.mod(Ni), z = i.div(Ni);
      const edge = x.equal(0).or(x.equal(this.N - 1)).or(z.equal(0)).or(z.equal(this.N - 1));
      const atC = (xx: any, zz: any) => curr.element(clamp(zz, 0, this.N - 1).mul(Ni).add(clamp(xx, 0, this.N - 1)));
      const h = curr.element(i), hp = prev.element(i);
      const lap = atC(x.add(1), z).add(atC(x.sub(1), z)).add(atC(x, z.add(1))).add(atC(x, z.sub(1))).sub(h.mul(4));
      const vel = h.sub(hp).mul(float(1).sub(this.dampU));
      let out = h.add(vel).add(lap.mul(this.c2dt2U));
      // 扰动注入
      const xf = x.toFloat(), zf = z.toFloat();
      Loop(this.dCountU, ({ i: di }: any) => {
        const dp = this.dPosU.element(di); // 格坐标
        const dv = this.dValU.element(di); // x=强度 y=半径
        const dist = vec2(xf.sub(dp.x), zf.sub(dp.y)).length();
        const w = float(1).sub(dist.div(dv.y)).max(0);
        out = out.add(dv.x.mul(w.mul(w)));
      });
      If(edge, () => {
        next.element(i).assign(
          reflectB ? atC(clamp(x, 1, this.N - 2), clamp(z, 1, this.N - 2)) : h.mul(0.5),
        );
      }).Else(() => { next.element(i).assign(out); });
      // 泡沫：波速度大处注入，整体衰减
      const f = foamBuf.element(i);
      f.assign(f.mul(0.985).add(vel.abs().mul(6).min(0.08)));
    })().compute(NN);

    const steps = [mkStep(a, b, c), mkStep(b, c, a), mkStep(c, a, b)];

    // 导出：把「当前」缓冲写入纹理（角色轮换由 CPU 侧传 ringIndex 决定 → 三个导出节点）
    const mkExport = (curr: any) => Fn(() => {
      const i = instanceIndex;
      const x = i.mod(Ni), z = i.div(Ni);
      textureStore(this.heightTex, ivec2(x, z), vec4(curr.element(i), 0, 0, 0));
      textureStore(this.foamTex, ivec2(x, z), vec4(foamBuf.element(i), 0, 0, 0));
    })().compute(NN);
    this.computes = { step: steps, export: [mkExport(c), mkExport(a), mkExport(b)] as any };
  }

  private ring = 0;

  /** 世界坐标处注入扰动。strength 正=抬升 负=压低（米），radius 米 */
  addDisturbance(worldX: number, worldZ: number, strength: number, radiusMeters: number) {
    if (this.disturbCount >= MAX_DISTURB) return;
    const gx = (worldX - this.originU.value.x) / this.dx + this.N / 2;
    const gz = (worldZ - this.originU.value.y) / this.dx + this.N / 2;
    if (gx < 1 || gx >= this.N - 1 || gz < 1 || gz >= this.N - 1) return;
    const k = this.disturbCount++;
    (this.dPosU as any).array[k].set(gx, gz);
    (this.dValU as any).array[k].set(strength, Math.max(radiusMeters / this.dx, 1.5));
  }

  /** 网格跟随目标（船/相机），整格吸附；简化：平移时旧波形留在原世界位置之外即被丢弃 */
  follow(targetX: number, targetZ: number) {
    const sx = Math.round(targetX / this.dx) * this.dx;
    const sz = Math.round(targetZ / this.dx) * this.dx;
    this.originU.value.set(sx, sz);
  }

  update(renderer: THREE.WebGPURenderer, dt: number) {
    // CFL 稳定性： c²dt²/dx² ≤ 0.4（固定子步 1/60s）
    const c2dt2 = (this.waveSpeed * this.waveSpeed * (1 / 60) ** 2) / (this.dx * this.dx);
    this.c2dt2U.value = Math.min(c2dt2, 0.4);
    this.dCountU.value = this.disturbCount;
    renderer.compute(this.computes.step[this.ring]);
    renderer.compute((this.computes.export as any)[this.ring]);
    this.ring = (this.ring + 1) % 3;
    this.disturbCount = 0;
  }

  /** 当前高度所在 instancedArray（供 HeightSampler 回读） */
  get currentBuffer() { return this.heightBuffers[(this.ring + 1) % 3]; } // step 后 next 即最新
}

function makeTex(N: number) {
  const t = new THREE.StorageTexture(N, N);
  t.type = THREE.HalfFloatType; t.format = THREE.RGBAFormat;
  t.magFilter = t.minFilter = THREE.LinearFilter;
  t.wrapS = t.wrapT = THREE.ClampToEdgeWrapping;
  return t;
}
```

```ts
// src/physics/HeightSampler.ts
import * as THREE from 'three/webgpu';

export function bilinearSample(arr: Float32Array, N: number, x: number, z: number): number {
  const cx = Math.min(Math.max(x, 0), N - 1), cz = Math.min(Math.max(z, 0), N - 1);
  const x0 = Math.floor(cx), z0 = Math.floor(cz);
  const x1 = Math.min(x0 + 1, N - 1), z1 = Math.min(z0 + 1, N - 1);
  const fx = cx - x0, fz = cz - z0;
  return arr[z0 * N + x0] * (1 - fx) * (1 - fz) + arr[z0 * N + x1] * fx * (1 - fz)
       + arr[z1 * N + x0] * (1 - fx) * fz + arr[z1 * N + x1] * fx * fz;
}

export interface GridMapping { originX: number; originZ: number; sizeMeters: number; N: number }

export function worldToGrid(wx: number, wz: number, m: GridMapping) {
  const half = (m.N - 1) / 2;
  return {
    x: ((wx - m.originX) / m.sizeMeters) * m.N + half,
    z: ((wz - m.originZ) / m.sizeMeters) * m.N + half,
  };
}

/** 维护一个 GPU float 缓冲的 CPU 镜像，异步回读（1 帧延迟），失败保留上帧 */
export class GpuHeightMirror {
  data: Float32Array;
  private inFlight = false;
  constructor(private renderer: THREE.WebGPURenderer, private bufferNode: any, readonly N: number) {
    this.data = new Float32Array(N * N);
  }
  /** 每帧调用一次；fire-and-forget */
  refresh() {
    if (this.inFlight) return;
    this.inFlight = true;
    // three webgpu: getArrayBufferAsync(StorageBufferAttribute)
    (this.renderer as any).getArrayBufferAsync(this.bufferNode.value)
      .then((ab: ArrayBuffer) => { this.data = new Float32Array(ab); })
      .catch(() => { /* 保留上帧数据 */ })
      .finally(() => { this.inFlight = false; });
  }
  sampleWorld(wx: number, wz: number, m: GridMapping): number {
    const g = worldToGrid(wx, wz, m);
    return bilinearSample(this.data, this.N, g.x, g.z);
  }
}

/** 组合采样器：FFT 级联高度 + 交互层高度 */
export class WaterHeightField {
  constructor(
    private layers: { mirror: GpuHeightMirror; mapping: () => GridMapping; tiling: boolean }[],
  ) {}
  height(wx: number, wz: number): number {
    let h = 0;
    for (const l of this.layers) {
      const m = l.mapping();
      if (l.tiling) {
        // FFT 平铺：世界坐标 wrap 进域
        const wrap = (v: number, s: number) => ((v % s) + s) % s;
        const g = { x: (wrap(wx, m.sizeMeters) / m.sizeMeters) * m.N, z: (wrap(wz, m.sizeMeters) / m.sizeMeters) * m.N };
        h += bilinearSample(l.mirror.data, l.mirror.N, g.x, g.z);
      } else {
        const g = worldToGrid(wx, wz, m);
        if (g.x < 0 || g.x > m.N - 1 || g.z < 0 || g.z > m.N - 1) continue;
        h += bilinearSample(l.mirror.data, l.mirror.N, g.x, g.z);
      }
    }
    return h;
  }
  refresh() { for (const l of this.layers) l.mirror.refresh(); }
}
```

```ts
// src/physics/RigidBody.ts
import { Vector3, Quaternion, Matrix3 } from 'three';

/** 盒近似惯量的半隐式欧拉刚体 */
export class RigidBody {
  position = new Vector3();
  quaternion = new Quaternion();
  velocity = new Vector3();
  angularVelocity = new Vector3();
  private forceAcc = new Vector3();
  private torqueAcc = new Vector3();
  private invInertia = new Matrix3();

  constructor(public mass: number, halfExtents: Vector3) {
    // 盒惯量 I = m/12 · (b²+c², a²+c², a²+b²)，a,b,c 为全尺寸
    const a = halfExtents.x * 2, b = halfExtents.y * 2, c = halfExtents.z * 2;
    const ix = (mass / 12) * (b * b + c * c), iy = (mass / 12) * (a * a + c * c), iz = (mass / 12) * (a * a + b * b);
    this.invInertia.set(1 / ix, 0, 0, 0, 1 / iy, 0, 0, 0, 1 / iz);
  }

  applyForce(f: Vector3) { this.forceAcc.add(f); }

  /** worldOffset: 作用点相对质心的世界系偏移 */
  applyForceAtPoint(f: Vector3, worldOffset: Vector3) {
    this.forceAcc.add(f);
    this.torqueAcc.add(new Vector3().crossVectors(worldOffset, f));
  }

  step(dt: number) {
    this.velocity.addScaledVector(this.forceAcc, dt / this.mass);
    this.position.addScaledVector(this.velocity, dt);
    // 简化：用体坐标惯量近似世界惯量（小角速度场景误差可接受）
    const angAcc = this.torqueAcc.clone().applyMatrix3(this.invInertia);
    this.angularVelocity.addScaledVector(angAcc, dt);
    const w = this.angularVelocity;
    const dq = new Quaternion(w.x * dt / 2, w.y * dt / 2, w.z * dt / 2, 0).multiply(this.quaternion);
    this.quaternion.set(this.quaternion.x + dq.x, this.quaternion.y + dq.y, this.quaternion.z + dq.z, this.quaternion.w + dq.w).normalize();
    this.forceAcc.set(0, 0, 0); this.torqueAcc.set(0, 0, 0);
  }
}
```

```ts
// src/physics/Floater.ts
import { Vector3 } from 'three';
import type { RigidBody } from './RigidBody';

const RHO = 1000, G = 9.81;

export interface FloaterOpts {
  points: Vector3[];        // 体坐标采样点（通常布在底面）
  crossSectionArea: number; // 水线总截面积 m²（所有点均分）
  maxDraft: number;         // 浸没深度饱和值（整体没入后浮力不再增加）
  linearDrag: number;       // N·s/m
  angularDrag: number;      // N·m·s
}

export type HeightFn = (wx: number, wz: number) => number;

export class Floater {
  constructor(private body: RigidBody, private o: FloaterOpts) {}

  /** 按当前水面高度施加浮力+阻尼到刚体（调用方随后自行加重力并 step） */
  applyForces(waterHeight: HeightFn, dt: number) {
    const perArea = this.o.crossSectionArea / this.o.points.length;
    let submergedAny = false;
    for (const p of this.o.points) {
      const wp = p.clone().applyQuaternion(this.body.quaternion).add(this.body.position);
      const wh = waterHeight(wp.x, wp.z);
      const depth = wh - wp.y;
      if (depth <= 0) continue;
      submergedAny = true;
      const d = Math.min(depth, this.o.maxDraft);
      const f = new Vector3(0, RHO * G * perArea * d, 0);
      this.body.applyForceAtPoint(f, wp.sub(this.body.position));
    }
    if (submergedAny) {
      this.body.applyForce(this.body.velocity.clone().multiplyScalar(-this.o.linearDrag));
      // 角阻尼：直接按比例衰减角速度（指数衰减的一阶近似）
      this.body.angularVelocity.multiplyScalar(Math.max(1 - this.o.angularDrag * dt, 0));
    }
  }
}
```

```ts
const MAX_ITEMS = 24;

export class Throwables {
  private items: Item[] = [];
  private ballGeo = new THREE.SphereGeometry(0.35, 24, 16);
  private boxGeo = new THREE.BoxGeometry(0.7, 0.7, 0.7);
  private ballMat = new THREE.MeshStandardNodeMaterial({ color: 0xd84a3a, roughness: 0.4 });

  spawn(kind: 'ball' | 'box', pos: Vector3, vel: Vector3) {

      this.scene.remove(old.mesh);

    const floater = new Floater(body, {
      points: pts,
      crossSectionArea: isBall ? Math.PI * 0.35 ** 2 : 0.49,
      maxDraft: 0.7, linearDrag: body.mass * 1.2, angularDrag: 1.5,
    });
```

```ts
// src/effects/SplashParticles.ts
import * as THREE from 'three/webgpu';
import { Vector3 } from 'three';
import {
  Fn, instancedArray, instanceIndex, uniform, float, vec3, vec4, If, hash,
} from 'three/tsl';

const COUNT = 8192;

export class SplashParticles {
  readonly points: THREE.Points;
  private positions = instancedArray(COUNT, 'vec3');
  private velocities = instancedArray(COUNT, 'vec3');
  private life = instancedArray(COUNT, 'float'); // ≤0 表示死亡
  private updateCompute: any;
  private emitCompute: any;
  private dtU = uniform(0);
  private emitPosU = uniform(new THREE.Vector3());
  private emitSpeedU = uniform(0);
  private emitSeedU = uniform(0);
  private emitStartU = uniform(0); // 环形分配起始下标
  private emitCountU = uniform(0);
  private cursor = 0;

  constructor(scene: THREE.Scene) {
    this.updateCompute = Fn(() => {
      const p = this.positions.element(instanceIndex);
      const v = this.velocities.element(instanceIndex);
      const l = this.life.element(instanceIndex);
      If(l.greaterThan(0), () => {
        v.y.subAssign(float(9.81).mul(this.dtU));
        p.addAssign(v.mul(this.dtU));
        l.subAssign(this.dtU);
      });
    })().compute(COUNT);

    this.emitCompute = Fn(() => {
      const rel = instanceIndex.toFloat();
      const active = instanceIndex.greaterThanEqual(this.emitStartU.toInt())
        .and(instanceIndex.lessThan(this.emitStartU.add(this.emitCountU).toInt()));
      If(active, () => {
        const p = this.positions.element(instanceIndex);
        const v = this.velocities.element(instanceIndex);
        const l = this.life.element(instanceIndex);
        const seed = rel.add(this.emitSeedU);
        const ang = hash(seed).mul(Math.PI * 2);
        const r = hash(seed.add(7)).mul(0.6);
        p.assign(vec3(
          this.emitPosU.x.add(ang.cos().mul(r)),
          this.emitPosU.y.add(0.1),
          this.emitPosU.z.add(ang.sin().mul(r)),
        ));
        const up = this.emitSpeedU.mul(0.35).add(hash(seed.add(13)).mul(1.5));
        v.assign(vec3(ang.cos().mul(hash(seed.add(3)).mul(2)), up, ang.sin().mul(hash(seed.add(5)).mul(2))));
        l.assign(hash(seed.add(11)).mul(0.6).add(0.5));
      });
    })().compute(COUNT);

    const mat = new THREE.PointsNodeMaterial({ transparent: true, depthWrite: false, blending: THREE.AdditiveBlending });
    mat.positionNode = this.positions.element(instanceIndex) as any;
    (mat as any).sizeNode = this.life.element(instanceIndex).max(0).mul(6).add(2);
    (mat as any).colorNode = vec4(0.9, 0.95, 1.0, this.life.element(instanceIndex).max(0).mul(0.8));
    const geo = new THREE.BufferGeometry();
    geo.setAttribute('position', new THREE.Float32BufferAttribute(new Float32Array(COUNT * 3), 3));
    this.points = new THREE.Points(geo, mat);
    this.points.frustumCulled = false;
    scene.add(this.points);
  }

  /** speed 越大喷溅越多越高 */
  burst(renderer: THREE.WebGPURenderer, pos: Vector3, speed: number) {
    const n = Math.min(Math.floor(speed * 18), 320);
    this.emitPosU.value.copy(pos);
    this.emitSpeedU.value = speed;
    this.emitSeedU.value = Math.random() * 1000;
    this.emitStartU.value = this.cursor;
    this.emitCountU.value = n;
    this.cursor = (this.cursor + n) % (COUNT - 400);
    renderer.compute(this.emitCompute);
  }

  update(renderer: THREE.WebGPURenderer, dt: number) {
    this.dtU.value = dt;
    renderer.compute(this.updateCompute);
  }
  dispose() { this.points.removeFromParent(); }
}
```

## 风险与应对（执行者参考）

| 风险 | 应对 |
|------|------|
| three TSL API 与计划代码签名不符 | 以 `node_modules/three` 内置类型与官方 `examples/webgpu_compute_*` 为准微调调用方式；**算法不改** |
| `getArrayBufferAsync` 不存在 | 改用 `readRenderTargetPixelsAsync` + 将高度写入 RenderTarget 的兜底路径 |
| FFT 输出花屏/棋盘 | 按序检查：蝶形表（已有 CPU 测试）→ (-1)^(x+z) 置换 → ping-pong 次数奇偶 |
| 浮力抖动 | 增加浮力采样点、提高 linearDrag、物理子步固定 1/60s |
| 半精度纹理不可线性过滤报错 | StorageTexture 改 FloatType 并请求 `float32-filterable` feature，或保持 HalfFloat（默认可过滤） |
