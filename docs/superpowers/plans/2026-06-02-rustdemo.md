```rust
// ============================================================
// Rust 特性演示项目
// 目标：通过可运行的代码，展示 Rust 相比 C++ 和 Python 的独特性
// 运行方法：cargo run
// ============================================================

use std::collections::HashMap;
use std::fs;
use std::sync::{Arc, Mutex};
use std::sync::mpsc;
use std::thread;

// ============================================================
// === 第1章：所有权与移动语义（Ownership & Move Semantics）===
// ============================================================
//
// 【Rust 的核心设计】
// 每一个值在 Rust 中都有且只有一个"所有者"（owner）。
// 当所有者离开作用域，值被自动释放——无需 GC，无需手动 free。
// 所有权可以"移动"（move）给另一个变量，移动后原变量失效。
//
// 【C++ 的做法】
//   std::string s1 = "hello";
//   std::string s2 = s1;  // 默认触发深拷贝，两份独立内存
//   // 用 std::move(s1) 可以移动，但编译器不阻止你之后继续用 s1
//   // 这是未定义行为（UB），运行时才会崩溃
//
// 【Python 的做法】
//   s1 = "hello"
//   s2 = s1  # 两个变量指向同一对象（引用语义），靠引用计数 GC 管理
//   # 没有"所有权"概念，GC 负责一切，但有性能开销和循环引用问题
fn demo_ownership() {
    let s1 = String::from("hello, Rust!");

    // move：s1 的所有权转移给 s2，s1 不再有效
    // 这在编译期强制执行——不是运行时检查，不会有任何性能开销
    let s2 = s1;

    // 下面这行会导致编译错误：value borrowed here after move
    // println!("{}", s1);  // ← 取消注释，看看编译器怎么报错

    println!("  所有权已转移，s1 失效，s2 = {}", s2);

    // Clone：显式深拷贝
    // C++ 的隐式深拷贝在 Rust 里必须显式写 .clone()
    // 这让性能开销"可见"——读代码时你能看到哪里有拷贝
    let s3 = s2.clone();
    println!("  显式克隆：s2 = {}, s3 = {}", s2, s3);

    // 整数类型（i32、f64 等）实现了 Copy trait，赋值时自动拷贝，不会 move
    let x = 42;
    let y = x; // x 没有 move，因为 i32 是 Copy 类型
    println!("  Copy 类型：x = {}, y = {}", x, y);

    // s2、s3 在这里离开函数作用域，内存自动释放，无需 free/delete
}

// ============================================================
// === 第2章：借用与引用（Borrowing & References）===
// ============================================================
//
// 【核心规则——借用检查器在编译期强制执行】
//   规则1：同一时间，可以有任意多个不可变借用（&T）
//   规则2：同一时间，只能有一个可变借用（&mut T）
//   规则3：可变借用存在期间，不能有任何不可变借用
// 这三条规则彻底消灭了数据竞争——在编译期，不是运行期。
//
// 【C++ 的做法】
//   void func(std::string& s) { ... }  // 引用，但编译器不检查别名
//   void bad()  {
//       std::string* p = new std::string("hi");
//       delete p;
//       std::cout << *p;  // use-after-free，编译通过，运行时崩溃
//   }
//
// 【Python 的做法】
//   def func(lst): lst.append(1)  # 一切都是引用，没有借用检查
//   # 同一对象可以被任意多个变量同时修改，靠程序员自律
fn demo_borrowing() {
    let mut text = String::from("Rust borrowing");

    // 不可变借用（&T）：可以同时存在多个
    let r1 = &text;
    let r2 = &text;
    println!("  不可变借用（可多个同时存在）：r1={}, r2={}", r1, r2);
    // r1 和 r2 在这里最后一次被使用，不可变借用结束

    // 可变借用（&mut T）：只能有一个，且此时不能有其他借用
    let r3 = &mut text;
    r3.push_str(" is safe!");
    println!("  可变借用修改后：{}", r3);
    // r3 借用结束

    // 函数参数中使用借用——不转移所有权，调用后仍可使用 text
    let length = calculate_length(&text); // 传引用，不移动所有权
    println!("  text 仍然有效，长度 = {}", length);

    // 【悬空引用】C++ 中函数返回局部变量的指针/引用会产生 UB：
    //   int* bad() { int x = 5; return &x; }  // 返回后 x 已被释放
    // Rust 编译器直接拒绝这种代码（生命周期检查），取消下面注释看报错：
    // fn dangling() -> &String { let s = String::from("x"); &s }
}

fn calculate_length(s: &String) -> usize {
    // s 是借用，函数结束后 s 释放，但原始数据不受影响
    s.len()
}

// ============================================================
// === 第3章：无 GC 的内存安全（Memory Safety without GC）===
// ============================================================
//
// 【Rust 的做法】
// 通过所有权系统在编译期确定每块内存的生命周期。
// 变量离开作用域时自动调用 Drop trait——类似 C++ 析构函数，
// 但 Rust 保证它一定被调用，不需要程序员记住。
//
// 【C++ 的做法】
//   // 需要手写析构函数，或用 RAII（unique_ptr / shared_ptr）
//   // 但 raw pointer 仍然随处可见，很容易忘记 delete
//   {
//       int* p = new int(42);  // 忘记 delete p → 内存泄漏
//   }                          // p 离开作用域，但内存没有释放！
//
// 【Python 的做法】
//   # 引用计数 GC 自动管理，但：
//   # 1. 循环引用不会被引用计数回收（a.next = b; b.prev = a → 永远不释放）
//   # 2. GC 暂停影响延迟敏感的程序
//   # 3. 无法精确控制释放时机（文件句柄、锁什么时候关闭？）

struct Resource {
    name: String,
}

// Drop trait：相当于 C++ 析构函数，但 Rust 保证一定被调用
// 不需要 virtual 关键字，不需要继承，编译器自动调用
impl Drop for Resource {
    fn drop(&mut self) {
        // 这里可以放清理逻辑：关闭文件、释放网络连接等
        println!("  【自动释放】资源 '{}' 被 Drop，内存已回收", self.name);
    }
}

fn demo_memory_safety() {
    println!("  --- 进入内层作用域 ---");
    {
        let _r = Resource { name: String::from("数据库连接") };
        println!("  正在使用资源...");
        // _r 在这里离开内层作用域，drop() 自动被调用
        // 不需要 delete，不需要 close()，不会泄漏
    }
    println!("  --- 离开内层作用域，资源已释放 ---");

    // Vec 同理：离开作用域时自动释放所有元素
    let numbers = vec![1u64, 2, 3, 4, 5];
    let sum: u64 = numbers.iter().sum();
    println!("  Vec sum = {}（函数结束时自动释放，无需 delete[]）", sum);
}

// ============================================================
// === 第4章：枚举与模式匹配（Enum & Pattern Matching）===
// ============================================================
//
// 【Rust 的枚举】是"代数数据类型"（Algebraic Data Type，ADT）
// 每个变体可以携带不同类型、不同数量的数据。
// match 必须穷举所有变体——漏写任何一个都是编译错误。
//
// 【C++ 的做法】
//   enum Status { Ok, Error };  // 只是整数常量，不能携带数据
//   switch(s) { case Ok: ...; } // 可以不写 default，遗漏分支不报错
//   // 要携带数据需要 union 或 std::variant（C++17），复杂且容易出错
//
// 【Python 的做法】
//   from enum import Enum
//   class Status(Enum): Ok=1; Error=2
//   # Python 3.10 有 match，但不强制穷举，遗漏分支只是静默忽略

// 带数据的枚举——每个变体携带不同类型的值
enum ProcessResult {
    Success(usize),       // 成功：携带处理的词数量
    IoError(String),      // IO 错误：携带错误描述
    EmptyFile(String),    // 空文件：携带文件路径
}

fn demo_enum_match() {
    let results = vec![
        ProcessResult::Success(42),
        ProcessResult::IoError(String::from("文件不存在")),
        ProcessResult::EmptyFile(String::from("data/empty.txt")),
    ];

    for result in results {
        // match 必须处理所有变体——漏写任意一个，编译器报错：
        // "non-exhaustive patterns: `EmptyFile(_)` not covered"
        match result {
            ProcessResult::Success(count) => {
                println!("  ✓ 处理成功，共 {} 个词", count);
            }
            ProcessResult::IoError(msg) => {
                println!("  ✗ IO 错误：{}", msg);
            }
            ProcessResult::EmptyFile(path) => {
                println!("  ⚠ 文件为空：{}", path);
            }
            // 如果注释掉上面任何一个 arm，编译器会提示遗漏
        }
    }

    // match 还能用于解构元组、结构体，比 if/else 链更简洁安全
    let pair = (1, -2);
    match pair {
        (x, y) if x > 0 && y > 0 => println!("  两个都是正数"),
        (x, _) if x > 0           => println!("  只有第一个是正数"),
        _                          => println!("  其他情况"),
    }
}

// ============================================================
// === 第5章：Result / Option 错误处理 ===
// ============================================================
//
// 【Option<T>】替代 null/nil/None 指针
//   Some(value) — 有值
//   None        — 无值
// 编译器强制你在使用 Option 之前处理 None 情况，
// 彻底消灭 NullPointerException 和 segfault。
//
// 【Result<T, E>】替代异常和错误码
//   Ok(value) — 成功，携带结果
//   Err(e)    — 失败，携带错误信息
// 与 C++/Python 异常不同，Result 是普通返回值，
// 调用方必须显式处理（或用 ? 传播），不能悄悄忽略。
//
// 【? 操作符】错误传播的语法糖
//   等价于：if let Err(e) = result { return Err(e.into()); }
//   C++ 中需要：if (!result) return result.error();（还不一定有这个语法）
//   Python 中需要：try/except，但异常可以悄悄穿透调用栈

fn read_file_words(path: &str) -> Result<HashMap<String, usize>, String> {
    // fs::read_to_string 返回 Result<String, io::Error>
    // .map_err() 把 io::Error 转成 String
    // ? 操作符：如果是 Err，立即从当前函数返回这个 Err
    let content = fs::read_to_string(path)
        .map_err(|e| format!("读取 {} 失败: {}", path, e))?;

    let mut freq: HashMap<String, usize> = HashMap::new();
    for word in content.split_whitespace() {
        // 转小写，去掉首尾标点
        let word = word.to_lowercase();
        let word = word.trim_matches(|c: char| !c.is_alphabetic());
        if !word.is_empty() {
            *freq.entry(word.to_string()).or_insert(0) += 1;
        }
    }

    Ok(freq) // 用 Ok 包裹成功结果返回
}

fn find_top_word(freq: &HashMap<String, usize>) -> Option<(&str, usize)> {
    // max_by_key 在 HashMap 为空时返回 None，非空时返回 Some
    // 调用方必须用 match 或 if let 处理 None，不能假设一定有值
    freq.iter()
        .max_by_key(|(_, count)| *count)
        .map(|(word, count)| (word.as_str(), *count))
}

fn demo_error_handling() {
    // Result 处理：match 显式处理成功和失败
    match read_file_words("data/file1.txt") {
        Ok(freq) => {
            println!("  成功读取 file1.txt，共 {} 个不同词", freq.len());
            // Option 处理：if let 处理"可能没有值"的情况
            if let Some((word, count)) = find_top_word(&freq) {
                println!("  出现最多的词：'{}' ({}次)", word, count);
            } else {
                println!("  文件为空");
            }
        }
        Err(e) => println!("  错误（第5章演示）：{}", e),
    }

    // unwrap_or：提供默认值，不需要 match
    let count = find_top_word(&HashMap::new())
        .map(|(_, c)| c)
        .unwrap_or(0);
    println!("  空 HashMap 的 top count（默认值）：{}", count);
}

// ============================================================
// === 第6章：无畏并发（Fearless Concurrency）===
// ============================================================
//
// 【Rust 的并发工具】
//   Arc<T>     — 原子引用计数，允许多线程共享一个值的所有权
//                类似 C++ shared_ptr，但天生线程安全（Atomic）
//   Mutex<T>   — 互斥锁，访问内部数据必须先 .lock()
//                忘记加锁？编译器拒绝编译，不是运行时 bug
//   mpsc channel — 多生产者单消费者消息队列，线程间传递数据
//
// 【C++ 的做法】
//   std::mutex mu;
//   void bad_thread() { shared_data++; }  // 忘记 lock：数据竞争，UB
//   // 编译器不报错，运行时可能崩溃，也可能悄悄产生错误结果
//
// 【Python 的做法】
//   import threading
//   # GIL（全局解释器锁）阻止多个线程同时执行 Python 字节码
//   # 结果：Python 的 threading 对 CPU 密集型任务几乎没有加速效果
//   # CPU 密集型要用 multiprocessing，但数据共享变得复杂
fn demo_concurrency() {
    // Arc<Mutex<Vec>> — 可以安全地在多线程之间共享的可变向量
    // Arc 管理所有权（多线程共享），Mutex 保证互斥访问
    let shared_log: Arc<Mutex<Vec<String>>> = Arc::new(Mutex::new(Vec::new()));

    // mpsc channel：多个线程发送（tx），一个线程接收（rx）
    let (tx, rx) = mpsc::channel::<String>();

    let mut handles = vec![];

    for i in 0..3 {
        let tx = tx.clone();           // 每个线程持有一个发送端副本
        let log = Arc::clone(&shared_log); // Arc::clone 只增加引用计数，不复制数据

        let handle = thread::spawn(move || {
            // move 把 tx、log、i 的所有权移入这个线程的闭包
            // 如果不用 Arc/Mutex，直接共享 shared_log，编译器会报错：
            //   "cannot be shared between threads safely"
            let msg = format!("线程 {} 完成", i);

            // lock() 返回 MutexGuard，MutexGuard 离开作用域时自动解锁
            // 不需要手动 unlock()，不会忘记解锁
            log.lock().unwrap().push(msg.clone());

            tx.send(msg).unwrap();
        });
        handles.push(handle);
    }

    drop(tx); // 关闭原始发送端；当所有 tx 副本都 drop 后，rx 的迭代结束

    // 接收所有线程发来的消息（channel 自动排队，不丢失）
    for msg in rx {
        println!("  channel 收到：{}", msg);
    }

    for handle in handles {
        handle.join().unwrap(); // 等待线程结束
    }

    println!("  共享日志（Arc<Mutex<Vec>>）：{:?}", shared_log.lock().unwrap());
}

    println!();

    // ── 主程序：多线程词频统计 ──────────────────────────────
    println!("══════════════════════════════════════════════");
    println!("【主程序】多线程并发词频统计");
    println!("══════════════════════════════════════════════");

    // 读取 data/ 目录（Result 错误处理——第5章）
    let entries = match fs::read_dir("data") {
        Ok(e) => e,
        Err(err) => {
            eprintln!("无法读取 data/ 目录: {}", err);
            return;
        }
    };

    // 收集所有 .txt 文件路径（Option + filter_map——第5章）
    let file_paths: Vec<String> = entries
        .filter_map(|entry| {
            let entry = entry.ok()?;
            let path = entry.path();
            if path.extension()?.to_str()? == "txt" {
                Some(path.to_string_lossy().into_owned())
            } else {
                None
            }
        })
        .collect();

    if file_paths.is_empty() {
        eprintln!("data/ 目录中没有 .txt 文件");
        return;
    }

    println!("找到 {} 个文件，开始并发处理...", file_paths.len());

    // Arc<Mutex<HashMap>>——线程安全的共享词频表（第6章）
    let total_freq: Arc<Mutex<HashMap<String, usize>>> =
        Arc::new(Mutex::new(HashMap::new()));

    let (tx, rx) = mpsc::channel::<Result<HashMap<String, usize>, String>>();

    // 为每个文件 spawn 一个线程（第6章）
    for path in file_paths {
        let tx = tx.clone();
        thread::spawn(move || {
            println!("  处理: {}", path);
            let result = read_file_words(&path); // Result 错误处理（第5章）
            tx.send(result).unwrap();
        });
    }
    drop(tx);

    // 主线程合并结果（Arc + Mutex——第6章）
    for result in rx {
        match result {
            Ok(freq) => {
                let mut total = total_freq.lock().unwrap();
                for (word, count) in freq {
                    *total.entry(word).or_insert(0) += count;
                }
            }
            Err(e) => eprintln!("  错误: {}", e),
        }
    }

    // 排序并输出 Top 10
    let total = total_freq.lock().unwrap();
    let mut word_counts: Vec<(&String, &usize)> = total.iter().collect();
    word_counts.sort_by(|a, b| b.1.cmp(a.1));

    println!();
    println!("Top 10 高频词：");
    println!("{:<20} {:>8}", "词语", "次数");
    println!("{}", "─".repeat(30));
    for (word, count) in word_counts.iter().take(10) {
        println!("{:<20} {:>8}", word, count);
    }
}
```
